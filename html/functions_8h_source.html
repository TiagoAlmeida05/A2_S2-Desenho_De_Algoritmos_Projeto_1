59void dijkstra(Graph<T> * g, const int &origin,bool walking) {
60 // Initialize the vertices
61 for(Vertex<T>* v : g->getVertexSet()) {
62 v->walking=walking;
63 if(!walking){
64 v->setDist(INF);
65 v->setPath(nullptr);
66 }
67 else {
68 v->setWDist(INF);
69 v->setWPath(nullptr);
70 }
71 }
72
73 Vertex<T>* s = g->findVertex(origin);
74 if (!s) {
75 cerr << "Error: Origin vertex not found!" << endl;
76 return;
77 }
78 if(!walking) s->setDist(0);
79 else s->setWDist(0);
80
81 MutablePriorityQueue<Vertex<T>> q;
82 q.insert(s);
83 while( ! q.empty() ) {
84 Vertex<T>* v = q.extractMin();
85
86 if (v == nullptr) {
87 cerr << "Error: Extracted null vertex from priority queue!" << endl;
88 return;
89 }
90 for(Edge<T>* e : v->getAdj()) {
91 if(e->isSelected()) continue;
92 if(v->isVisited()) continue;
93 double oldDist = walking ? e->getDest()->getWDist() : e->getDest()->getDist();
94
95 if (relax(e,walking)) {
96 if (oldDist == INF) {
97 q.insert(e->getDest());
98 }
99 else {
100 q.decreaseKey(e->getDest());
101 }
102 }
103 }
